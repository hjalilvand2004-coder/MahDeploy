 ۱) پیش‌نیازها
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  ✔ ویندوز ۱۰ / ۱۱ یا ویندوز سرور ۲۰۱۶ به بالا (۶۴ بیتی)
  ✔ هیچ چیز دیگری! نه پایتون، نه Node.js، نه PyCharm، نه دیتابیس —
    تمام ران‌تایم‌ها داخل همین پوشه هستند (کاملاً Portable).

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
 ۲) اجرای سریع (Quick Start)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  ۱. کل پوشه MahDeploy را در مسیر دلخواه کپی کنید
     (پیشنهاد: C:\MahDeploy — از مسیرهای دارای فاصله یا فارسی پرهیز کنید)
  ۲. روی فایل  Start-MahDeploy.bat  دوبار کلیک کنید.
  ۳. چند ثانیه بعد مرورگر به‌صورت خودکار باز می‌شود:
         http://localhost:3000
  ۴. وارد شوید:
         نام کاربری: admin
         رمز عبور  : admin1234
     ⚠ بلافاصله از منوی تنظیمات، رمز عبور را عوض کنید!

  ▸ اجرا روی پورت دیگر (مثلاً ۸۰۸۰):
         Start-MahDeploy.bat 8080

  ▸ توقف پنل:
         فایل  Stop-MahDeploy.bat  را اجرا کنید یا پنجره را ببندید.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
 ۳) اجرا به‌صورت سرویس ویندوز (شروع خودکار همراه بوت سرور)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  PowerShell را با دسترسی Administrator باز کنید و در پوشه MahDeploy:

      powershell -ExecutionPolicy Bypass -File .\service-install.ps1

  پنل به‌صورت سرویس ویندوز با نام  MahDeploy Panel  ثبت و اجرا می‌شود
  (در صورت کرش، خودکار دوباره بالا می‌آید).

  حذف سرویس:
      powershell -ExecutionPolicy Bypass -File .\service-uninstall.ps1

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
 ۴) دسترسی از سایر کامپیوترها (شبکه / اینترنت)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  پنل به‌صورت پیش‌فرض روی همه کارت‌های شبکه گوش می‌دهد؛ کافی است در
  فایروال ویندوز پورت را باز کنید (PowerShell با دسترسی ادمین):

      New-NetFirewallRule -DisplayName "MahDeploy Panel" -Direction Inbound `
        -Protocol TCP -LocalPort 3000 -Action Allow

  سپس از سایر سیستم‌ها:   http://IP-سرور:3000

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
 ۵) ساختار پوشه‌ها
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  MahDeploy\
   ├─ Start-MahDeploy.bat      ← اجرای پنل (دوبار کلیک)
   ├─ Stop-MahDeploy.bat       ← توقف پنل
   ├─ MahDeploy-Guide-fa.pdf  ← 📖 راهنمای جامع کاربر (PDF مصور)
   ├─ service-install.ps1      ← ثبت سرویس ویندوز
   ├─ service-uninstall.ps1    ← حذف سرویس ویندوز
   ├─ service.xml              ← تنظیمات سرویس (WinSW)
   ├─ runtime\node.exe         ← ران‌تایم Node.js (داخل‌بسته)
   ├─ app\                     ← برنامه پنل (Next.js استاندالون)
   ├─ data\mahdeploy.db        ← دیتابیس پنل (پشتیبان‌گیری فقط همین فایل!)
   └─ logs\                    ← لاگ‌های اجرا

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
 ۶) پشتیبان‌گیری و انتقال
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  • پشتیبان‌گیری کامل = کپی کل پوشه MahDeploy (شامل data\mahdeploy.db)
  • انتقال به سرور جدید = کپی همان پوشه؛ همه‌چیز (کاربران، استقرارها،
    لاگ‌ها) حفظ می‌شود.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
 ۷) حل مشکلات رایج
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  ▸ «پورت اشغال است»
      با پورت دیگر اجرا کنید:  Start-MahDeploy.bat 3100
  ▸ پنل بالا نمی‌آید
      فایل لاگ در پوشه logs را ببینید؛ مطمئن شوید آنتی‌ویروس فایل
      node.exe را قرنطینه نکرده باشد (در صورت نیاز مسیر را Exclude کنید).
  ▸ مرورگر خودکار باز نشد
      دستی به آدرس  http://localhost:3000  بروید.
  ▸ رمز عبور را فراموش کردم
      فایل data\mahdeploy.db را حذف کنید؛ پنل با رمز پیش‌فرض
      admin1234 از نو راه‌اندازی می‌شود (تمام داده‌ها پاک می‌شود).
  ▸ کاربر «Administrator» بودن PowerShell برای اجرای اسکریپت‌های استقرار
    الزامی است؛ در غیر این صورت پنل فقط در حالت نمایش کار می‌کند.

 ۸) امکانات سامانه
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  ✔ داشبورد سلامت سرور (CPU / RAM / دیسک به‌صورت زنده)
  ✔ نقشه پورت‌های اشغال‌شده سرور
  ✔ پیش‌بررسی کامل قبل از استقرار (DNS دامنه، آزاد بودن پورت،
    اعتبارسنجی مسیر گواهی SSL و مسیر کد)
  ✔ استقرار ۵ مرحله‌ای با اسکریپت‌های PowerShell استاندارد و ترمینال زنده
  ✔ ساخت خودکار کانفیگ Nginx (HTTPS + Reverse Proxy)
  ✔ ثبت بک‌اند FastAPI به‌عنوان سرویس ویندوز با WinSW (شروع خودکار)
  ✔ مدیریت سیستم‌های مستقرشده: شروع / توقف / ری‌استارت / حذف
  ✔ تلاش دوبارهٔ مرحلهٔ ناموفق بدون بازسازی پلن (ادامه از همان مرحله)
  ✔ رابط کاربری فارسی (RTL) با طراحی شیشه‌ای مدرن
  ✔ راهنمای درون‌برنامه‌ای: هر بخش پنل دکمه «راهنما» دارد + مستند PDF مصور فارسی
